'use strict';
/* Moonvale HUD: ornate portrait-first layout (portrait frame + HP/MP/XP, currency,
   round minimap with zone plate and clock, quest panel, joystick, radial skill cluster,
   chat line and a bottom navigation bar). Reuses the game's own elements and ids,
   so all existing game logic keeps updating them. */
(()=>{
 const field=$('field'),footer=document.querySelector('footer');if(!field||!footer)return;
 document.body.classList.add('mv-hud');
 const el=(tag,id,cls,html='')=>{const e=document.createElement(tag);if(id)e.id=id;if(cls)e.className=cls;e.innerHTML=html;return e;};
 // Zone banner.
 if(!$('zoneBanner'))field.appendChild(el('div','zoneBanner','','<i></i><b></b><small></small>'));
 // Portrait + XP inside the player frame.
 const frame=$('playerFrame');
 const portrait=el('div','mvPortrait','mv-portrait','<canvas width="128" height="128"></canvas><span id="mvLevel">1</span>');
 frame.prepend(portrait);
 const xp=document.querySelector('footer .progress');if(xp){xp.classList.add('mv-xp');frame.appendChild(xp);}
 // Currency pills.
 const cur=el('div','mvCurrency','mv-currency','<span class="mv-pill gold"><i></i><b id="mvGold">0</b></span><span class="mv-pill gem"><i></i><b id="mvPower">0</b></span>');field.appendChild(cur);
 // Minimap plate + clock.
 const map=document.querySelector('.map-wrap');if(map){map.appendChild(el('div','mvMapPlate','mv-map-plate','<b id="mvZone"></b><small id="moonClock"></small>'));map.prepend(el('i','','mv-compass','N'));}
 // Radial skill cluster: move the hotbar and auto attack into the field.
 const radial=el('div','mvRadial','mv-radial');field.appendChild(radial);
 const hot=$('hotbar'),auto=$('autoBtn');if(hot)radial.appendChild(hot);if(auto)radial.appendChild(auto);
 // Chat / combat line.
 const chat=$('combatLog');if(chat)chat.classList.add('mv-chat');
 // Bottom navigation.
 const nav=el('nav','mvNav','mv-nav');
 const items=[['hero','Character','classBtn'],['bag','Inventory',()=>typeof openInventory==='function'&&openInventory()],['skills','Skills','menuSkills'],['quests','Quests',()=>typeof openJournal==='function'&&openJournal()],['codex','Codex','helpBtn'],['menu','Menu','adventureMenu']];
 for(const [icon,label,act] of items){const b=el('button','','mv-nav-'+icon,'<i></i><span>'+label+'</span>');b.onclick=()=>{if(typeof act==='function')return act();const t=$(act);if(t)t.click();else if(act==='menuSkills'){$('adventureMenu')?.click();setTimeout(()=>$('menuSkills')?.click(),30);}};nav.appendChild(b);}
 footer.appendChild(nav);
 // Portrait painter: the hero's own front frame, cropped to the head and shoulders.
 let lastKey='';
 function paintPortrait(){const f=heroFrame?.(p.cls,'front');if(!f)return;const src=(f.novice&&typeof jointSource==='function')?(()=>{try{const c=jointSource(f,true);return {img:c,sx:0,sy:0,sw:c.width,sh:c.height};}catch(e){return f;}})():f;
  const key=p.cls+'|'+src.sw+'|'+src.sh+'|'+(src.img?.width||0);if(key===lastKey)return;lastKey=key;
  const c=portrait.querySelector('canvas'),x=c.getContext('2d');x.clearRect(0,0,128,128);
  const cropH=src.sh*.36,cropW=Math.min(src.sw,cropH);x.imageSmoothingQuality='high';
  x.drawImage(src.img,src.sx+(src.sw-cropW)/2,src.sy,cropW,cropH,4,6,120,120*cropH/cropW);}
 let tick=0;
 function sync(){tick++;
  $('mvLevel').textContent=p.level;$('mvGold').textContent=Number(p.gold||0).toLocaleString();$('mvPower').textContent=$('power')?.textContent||'';
  $('mvZone').textContent=(typeof zones!=='undefined'&&zones[story.zone]?.name)||'';
  if(tick%30===1)paintPortrait();
  requestAnimationFrame(sync);}
 requestAnimationFrame(sync);
 // Layout changed: let the game recompute the canvas size.
 setTimeout(()=>window.dispatchEvent(new Event('resize')),50);
})();
